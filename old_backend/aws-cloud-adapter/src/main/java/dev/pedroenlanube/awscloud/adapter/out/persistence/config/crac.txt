    /**
     * This method runs in the AWS cloud DURING the deployment phase,
     * milliseconds before AWS freezes the Lambda’s memory.
     */
    @Override
    public void beforeCheckpoint(Context<? extends Resource> context) throws Exception {
        log.info("[CRaC] beforeCheckpoint: Forcing socket opening and TLS tunnelling with DynamoDB...");
        try {
            // CWe create a fake key that we know does not exist
            Key dummyKey = Key.builder()
                    .partitionValue("WARMUP#DUMMY")
                    .sortValue("WARMUP#DUMMY")
                    .build();
            // We have launched a formal request to compel the AWS HTTP Client to negotiate
            // SSL and open the connection pool.
            userTable.getItem(dummyKey);

        } catch (Exception e) {
            // We catch EVERYTHING. If the request fails (due to permissions or whatever), we don’t want the Lambda
            // function to abort. The network tunnel will still have been attempted, which is what matters to us.
            log.warn("[CRaC] Expected error during priming (ignored): {}", e.getMessage());
        }
        log.info("[CRaC] beforeCheckpoint: Hot connections, ready to freeze!");
    }

    /**
     * This method runs when a request from Cognito arrives and
     * AWS unfreezes the memory to handle it.
     */
    @Override
    public void afterRestore(Context<? extends Resource> context) throws Exception {
        log.info("[CRaC] afterRestore: Lambda unfrozen. The connections with DynamoDB are already established.");
    }
